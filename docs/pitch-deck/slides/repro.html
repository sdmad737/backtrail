adb shell am instrument -w -e class com.backtrail.app.bench.GemmaDeviceBench#describe -e backend NPU -e model gemma-4-E2B-it_Google_Tensor_G6.litertlm com.backtrail.app.test/androidx.test.runner.AndroidJUnitRunner
